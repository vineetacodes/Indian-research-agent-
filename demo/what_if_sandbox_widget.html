def calculate_what_if(cap_mt=10, util_pct=70.0, grid_tariff=7.80, solar_sub_pct=0.75, pmksy_pct=0.35):
    cold_capex = cap_mt * 120_000
    solar_capex = 5.0 * 45_000  # 5 kW array
    
    net_capex = (cold_capex * (1 - pmksy_pct)) + (solar_capex * (1 - solar_sub_pct))
    per_farmer_outlay = net_capex / 10.0
    
    annual_crop_mt = cap_mt * 4 * (util_pct / 100.0)
    spoilage_savings = annual_crop_mt * 0.12 * 25_000
    price_arbitrage = annual_crop_mt * 25_000 * 0.18
    solar_kwh_savings = (5.0 * 4.2 * 270) * grid_tariff
    amc_opex = (cold_capex + solar_capex) * 0.03
    
    net_annual_profit = (spoilage_savings + price_arbitrage + solar_kwh_savings) - amc_opex
    payback_years = net_capex / max(1000.0, net_annual_profit)
    irr_pct = max(6.0, (net_annual_profit / net_capex) * 100.0 - 4.0)
    
    return {
        "net_capex": net_capex,
        "per_farmer_outlay": per_farmer_outlay,
        "net_annual_profit": net_annual_profit,
        "payback_years": payback_years,
        "irr_pct": irr_pct
    }
